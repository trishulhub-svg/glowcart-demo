    41→function ShipmentStatusBadge({ status }: { status: Shipment['status'] }) {
    42→  const config: Record<Shipment['status'], { label: string; className: string }> = {
    43→    picked_up: { label: 'Picked Up', className: 'bg-violet-100 text-violet-800 border-violet-200 hover:bg-violet-100' },
    44→    in_transit: { label: 'In Transit', className: 'bg-sky-100 text-sky-800 border-sky-200 hover:bg-sky-100' },
    45→    out_for_delivery: { label: 'Out for Delivery', className: 'bg-amber-100 text-amber-800 border-amber-200 hover:bg-amber-100' },
    46→    delivered: { label: 'Delivered', className: 'bg-emerald-100 text-emerald-800 border-emerald-200 hover:bg-emerald-100' },
    47→  };
    48→  const c = config[status];
    49→  return <Badge variant="outline" className={c.className}>{c.label}</Badge>;

    53→function formatDate(iso: string) {
    54→  return new Date(iso).toLocaleDateString('en-IN', {
    55→    day: '2-digit',
    56→    month: 'short',
    57→    year: 'numeric',
    58→  });
    59→}

    61→function formatDateShort(iso: string) {

    65→  });

   615→// ─── Delivery History ──────────────────────────────────────────────────────
   616→function DeliveryHistory({
   617→  completedShipments,
   618→  completedOrders,
   619→  orders,
   620→}: {
   621→  completedShipments: Shipment[];
   622→  completedOrders: Order[];
   623→  orders: Order[];
   624→}) {
   625→  const [filterPeriod, setFilterPeriod] = useState<'all' | 'week' | 'month'>('all');
   626→
   627→  // Combine shipments and delivered orders for history
   628→  const historyItems = useMemo(() => {
   629→    const deliveredOrders = orders.filter((o) => o.status === 'delivered');
   630→    return deliveredOrders.map((order) => {
   631→      const shipment = completedShipments.find((s) => s.orderId === order.id);
   632→      return {
   633→        orderId: order.id,
   634→        customer: order.customerName,
   635→        date: order.updatedAt,
   636→        status: shipment?.status || ('delivered' as const),
   637→        address: order.shippingAddress,
   638→        total: order.total,
   639→      };
   640→    });
   641→  }, [orders, completedShipments]);
   642→
   643→  const filteredHistory = useMemo(() => {
   644→    if (filterPeriod === 'all') return historyItems;
   645→    const now = new Date();
   646→    const cutoff = filterPeriod === 'week'
   647→      ? new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   648→      : new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
   649→    return historyItems.filter((item) => new Date(item.date) >= cutoff);
   650→  }, [historyItems, filterPeriod]);
   651→
   652→  return (
   653→    <div className="space-y-5">
   654→      <div>
   655→        <h1 className="text-xl font-bold tracking-tight">Delivery History</h1>
   656→        <p className="text-muted-foreground text-sm mt-0.5">
   657→          {historyItems.length} completed deliver{historyItems.length !== 1 ? 'ies' : 'y'}
   658→        </p>
   659→      </div>
   660→
   661→      {/* Date Filter */}
   662→      <div className="flex items-center gap-1.5 sm:gap-2 overflow-x-auto pb-1">
   663→        {(['all', 'week', 'month'] as const).map((period) => (
   664→          <Button
   665→            key={period}
   666→            size="sm"
   667→            variant={filterPeriod === period ? 'default' : 'outline'}
   668→            className={`text-xs min-h-[44px] whitespace-nowrap ${
   669→              filterPeriod === period
   670→                ? 'bg-rose-500 hover:bg-rose-600 text-white'
   671→                : 'border-rose-200 text-rose-700 hover:bg-rose-50'
   672→            }`}
   673→            onClick={() => setFilterPeriod(period)}
   674→          >
   675→            <Calendar className="h-3 w-3 mr-1" />
   676→            {period === 'all' ? 'All Time' : period === 'week' ? 'This Week' : 'This Month'}
   677→          </Button>
   678→        ))}
   679→      </div>
   680→
   681→      {/* History List */}
   682→      {filteredHistory.length === 0 ? (
   683→        <Card className="border-0 shadow-md">
   684→          <CardContent className="flex flex-col items-center justify-center py-14">
   685→            <History className="h-10 w-10 text-slate-300 mb-2" />
   686→            <p className="text-sm text-muted-foreground">No delivery history found</p>
   687→          </CardContent>
   688→        </Card>
   689→      ) : (
   690→        <div className="space-y-3">
   691→          {filteredHistory.map((item, i) => (
   692→            <motion.div
   693→              key={item.orderId}
   694→              initial={{ opacity: 0, x: -8 }}
   695→              animate={{ opacity: 1, x: 0 }}
   696→              transition={{ delay: i * 0.04 }}
   697→            >
   698→              <Card className="border-0 shadow-md hover:shadow-lg transition-shadow overflow-hidden">
   699→                <CardContent className="p-3 sm:p-4">
   700→                  <div className="flex items-start justify-between">
   701→                    <div className="flex items-center gap-2 sm:gap-3">
   702→                      <div className="flex h-9 w-9 sm:h-10 sm:w-10 items-center justify-center rounded-xl bg-emerald-100 text-emerald-600">
   703→                        <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5" />
   704→                      </div>
   705→                      <div>
   706→                        <p className="font-mono text-sm font-bold text-rose-700">{item.orderId}</p>
   707→                        <p className="text-sm font-medium">{item.customer}</p>
   708→                      </div>
   709→                    </div>
   710→                    <div className="text-right">
   711→                      <Badge variant="outline" className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[10px]">
   712→                        Delivered
   713→                      </Badge>
   714→                      <p className="text-xs text-muted-foreground mt-1">{formatDate(item.date)}</p>
   715→                    </div>
   716→                  </div>
   717→
   718→                  <Separator className="my-3 bg-rose-100/50" />
   719→
   720→                  <div className="flex items-center justify-between text-sm">
   721→                    <div className="flex items-center gap-1 text-muted-foreground">
   722→                      <MapPin className="h-3.5 w-3.5" />
   723→                      <span className="truncate max-w-[120px] sm:max-w-[180px] text-xs">{item.address}</span>
   724→                    </div>
   725→                    <span className="font-semibold text-xs">{formatCurrency(item.total)}</span>
   726→                  </div>
   727→                </CardContent>
   728→              </Card>
   729→            </motion.div>
   730→          ))}
   731→        </div>
   732→      )}
   733→    </div>
   734→  );
   735→}
