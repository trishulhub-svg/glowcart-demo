   626→   626→
   627→   627→  // Combine shipments and delivered orders for history
   628→   628→  const historyItems = useMemo(() => {
   629→   629→    const deliveredOrders = orders.filter((o) => o.status === 'delivered');
   630→   630→    return deliveredOrders.map((order) => {
   631→   631→      const shipment = completedShipments.find((s) => s.orderId === order.id);

   633→   633→        orderId: order.id,
   634→   634→        customer: order.customerName,

   636→   636→        status: shipment?.status || ('delivered' as const),

   638→   638→        total: order.total,
   639→   639→      };
   640→   640→    });
